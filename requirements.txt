# WyrmHex v1.2.0 - libraries to install
# Install with:  pip install -r requirements.txt
# (with the virtual environment active: see step 4 and 5 in the README)
#
# Pillow creates and saves the map pictures.
# 11.3 is the first version made for Python 3.14.
Pillow>=11.3
#
# asciimatics draws the menus you move through with the arrow keys,
# and the editor that shows the map in the terminal. Without it the
# program still works: it asks for the option's number, and the editor
# asks questions.
asciimatics>=1.15
#
# fpdf2 writes the PDF with the contents of the land (encounter tables,
# history, hexes). Only needed if you ask for the contents: without it
# the map is made anyway, and the contents are saved as JSON only.
fpdf2>=2.7
